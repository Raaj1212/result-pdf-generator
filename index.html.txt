// --- UTILITY FUNCTIONS ---
function esc(v){ return String(v??'').replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;'); }

function fmt(v){
  if(v === null || v === undefined || v === '') return '-';
  if(typeof v === 'number') return Number.isInteger(v) ? String(v) : v.toFixed(2);
  const s = String(v).trim(); const n = Number(s);
  return (!isNaN(n) && s !== '') ? (Number.isInteger(n) ? String(n) : n.toFixed(2)) : esc(s);
}

function formatSubhead(text){
  if(!text) return '-';
  return esc(text).replace(/(\s+)(\d+(\.\d+)?)$/, '<br>$2');
}

function checkAuthSession(){
  if(sessionStorage.getItem('portal_auth_token') === 'verified_access'){
    loginOverlay.style.display = 'none';
  } else {
    loginOverlay.style.display = 'flex';
    passInput.focus();
  }
}
function doLogin(){
  if(VALID_PASSWORDS.includes(passInput.value.trim())){
    sessionStorage.setItem('portal_auth_token', 'verified_access');
    loginOverlay.style.display = 'none'; loginError.style.display = 'none'; passInput.value = '';
    loadExamSheetsDropdown();
  } else {
    loginError.style.display = 'block'; passInput.value = ''; passInput.focus();
  }
}

});

function renderAdminSheetsList(){

  });

    btn.addEventListener('click', function(){ deleteSheetFromList(this.getAttribute('data-id')); });
  });

// Universal Parsing Engine
function parseRawSheets(sheetObjects, forceMode = 'auto'){
  const models = [];
  for(const item of sheetObjects){
    const sheetName = item.name; const rows = item.rows;
    if(!rows || rows.length < 4) continue;
    let isSA = false; let saSubjectRow = -1, saHeaderRow = -1;
    for(let r=0; r<Math.min(12, rows.length); r++){
      const text = (rows[r]||[]).map(v=>String(v??'').toUpperCase()).join('|');
      if((text.includes('POST MID') || text.includes('NOTE BOOK') || text.includes('ANNUAL')) && r > 0){
        isSA = true; saSubjectRow = r - 1; saHeaderRow = r; break;
      }
    }
    if(forceMode === 'SA') isSA = true; else if(forceMode === 'FA') isSA = false;

    if(isSA){
      const rowSubj = rows[saSubjectRow >= 0 ? saSubjectRow : 2] || [];
      const rowHead = rows[saHeaderRow >= 0 ? saHeaderRow : 3] || [];
      let rnCol = 0, nameCol = 1;
      for(let c=0; c<rowHead.length; c++){
        const s = String(rowHead[c]??'').toUpperCase();
        if(s === 'RN' || s === 'ROLL') rnCol = c;
        if(s === 'NAME' || s.includes('STUDENT NAME')) nameCol = c;
      }
      const rawSubjectIndices = [];
      for(let c=0; c<rowSubj.length; c++){
        const v = rowSubj[c]; const hText = String(rowHead[c]??'').toUpperCase().replace(/\s+/g,''); const sText = String(v??'').toUpperCase().replace(/\s+/g,'');
        if(sText.includes('GRAND') || sText.includes('PERCENTAGE') || sText.includes('PER(%)') || sText === 'TOTAL' || hText.includes('GRAND') || hText.includes('PERCENTAGE') || hText.includes('PER(%)') || hText === 'TOTAL') break;
        if(v!==null && v!=='' && !['GRAND TOTAL','PER(%)','PER (%)','PERCENTAGE','RN','ROLL','NAME','STUDENT NAME'].includes(String(v).trim().toUpperCase()) && c !== rnCol && c !== nameCol){
          rawSubjectIndices.push({name: String(v).trim(), start: c});
        }
      }
      if(rawSubjectIndices.length > 0){
        const rawSubjects = []; let lastSubjectEndCol = 0;
        for(let i=0; i<rawSubjectIndices.length; i++){
          const curr = rawSubjectIndices[i]; const nextStart = (i + 1 < rawSubjectIndices.length) ? rawSubjectIndices[i+1].start : curr.start + 5;
          const colCount = Math.max(1, nextStart - curr.start); lastSubjectEndCol = curr.start + colCount;
          const subheads = [];
          for(let off=0; off<colCount; off++){ const sh = rowHead[curr.start + off]; subheads.push(sh ? String(sh).trim().replace(/\s+/g,' ') : `Sub ${off+1}`); }
          rawSubjects.push({name: curr.name, start: curr.start, colCount: colCount, subheads: subheads});
        }
        const grandCol = lastSubjectEndCol; const perCol = lastSubjectEndCol + 1;
        const startRow = (saHeaderRow >= 0 ? saHeaderRow : 3) + 1;
        const activeSubjects = [];
        rawSubjects.forEach(s => {
          let hasAnyMark = false;
          for(let r=startRow; r<rows.length; r++){
            const name = rows[r]?.[nameCol]; const rn = rows[r]?.[rnCol];
            if((name && String(name).trim() !== '') || (rn && String(rn).trim() !== '')){
              const block = (rows[r] || []).slice(s.start, s.start + s.colCount);
              if(block.some(v => v !== null && v !== undefined && String(v).trim() !== '')){ hasAnyMark = true; break; }
            }
          }
          if(hasAnyMark) activeSubjects.push(s);
        });
        const finalSubjects = (activeSubjects.length > 0) ? activeSubjects : rawSubjects;
        const students = [];
        for(let r=startRow; r<rows.length; r++){
          const rn = rows[r]?.[rnCol]; const name = rows[r]?.[nameCol];
          const blocks = finalSubjects.map(s => (rows[r] || []).slice(s.start, s.start + s.colCount));
          const hasValidRoll = (typeof rn === 'number' || /^\d+$/.test(String(rn ?? '').trim()));
          const hasName = (name && String(name).trim() !== '' && !['TOTAL', 'AVERAGE', 'PERCENTAGE', 'GRADE'].includes(String(name).trim().toUpperCase()));
          const hasMarks = blocks.some(b => b.some(v => v !== null && v !== undefined && String(v).trim() !== ''));
          if(hasName || (hasValidRoll && hasMarks)){
            students.push({ rn: rn || (students.length + 1), name: hasName ? String(name).trim() : '-', blocks: blocks, grand: rows[r][grandCol] ?? null, per: rows[r][perCol] ?? null });
          }
        }
        const title = (rows[0]||[]).find(v=>typeof v==='string' && v.trim().length > 3) || 'BRAHMADEV PUBLIC SCHOOL , SITAMARHI';
        const subtitleRaw = (rows[1]||[]).find(v=>typeof v==='string' && v.trim().length > 3) || 'Session : 2026-27 ; MARKS STATEMENT (SA 1)';
        const subtitleBase = subtitleRaw.replace(/Class\s*:\s*.*$/i, '').trim().replace(/;$/, '').trim();
        models.push({ examType: 'SA', sheetName, title: title.trim(), subtitleBase, subjects: finalSubjects, students });
      }
    } else {
      let faHeaderRow = -1;
      for(let r=0; r<Math.min(10, rows.length); r++){
        const rowStr = (rows[r]||[]).map(v=>String(v??'').toUpperCase()).join('|');
        if((rowStr.includes('ROLL') || rowStr.includes('NAME') || rowStr.includes('RN')) && (rowStr.includes('TOTAL') || rowStr.includes('MATH') || rowStr.includes('ENGLISH') || rowStr.includes('OUT OF 20'))){
          faHeaderRow = r; break;
        }
      }
      if(faHeaderRow < 0) faHeaderRow = 3;
      const hRow = rows[faHeaderRow] || [];
      let rnCol = -1, nameCol = -1; const rawSubjects = [];
      for(let c=0; c<hRow.length; c++){
        const s = String(hRow[c]??'').trim().toUpperCase(); if(!s) continue;
        if(s === 'ROLL' || s === 'RN' || s === 'R.NO' || s === 'ROLL NO' || s === 'ROLL NO.' || s === 'SL'){ rnCol = c; }
        else if(s.includes('NAME')){ nameCol = c; }
        else if(s === 'TOTAL' || s === 'GRAND TOTAL' || s === 'TOT' || s.includes('PER') || s.includes('%') || s.includes('PERCENT') || s.includes('F.MARKS') || s.includes('FULL MARKS')){ break; }
        else { rawSubjects.push({ name: String(hRow[c]).trim().replace(/\(\s*OUT OF\s*20\s*\)/i, '').trim(), col: c }); }
      }
      if(rnCol === -1) rnCol = 0; if(nameCol === -1) nameCol = 1;
      const lastSubjCol = rawSubjects.length > 0 ? rawSubjects[rawSubjects.length - 1].col : 1;
      const totalCol = lastSubjCol + 1; const perCol = lastSubjCol + 2;
      const activeSubjects = [];
      rawSubjects.forEach(s => {
        let hasAnyMark = false;
        for(let r=faHeaderRow+1; r<rows.length; r++){
          const name = rows[r]?.[nameCol]; const rn = rows[r]?.[rnCol];
          if((name && String(name).trim() !== '') || (rn && String(rn).trim() !== '')){
            const val = rows[r]?.[s.col]; if(val !== null && val !== undefined && String(val).trim() !== ''){ hasAnyMark = true; break; }
          }
        }
        if(hasAnyMark) activeSubjects.push(s);
      });
      const finalSubjects = (activeSubjects.length > 0) ? activeSubjects : rawSubjects;
      const students = [];
      for(let r=faHeaderRow+1; r<rows.length; r++){
        const rn = rows[r]?.[rnCol]; const name = rows[r]?.[nameCol];
        const marks = finalSubjects.map(s => rows[r]?.[s.col] ?? null);
        const hasValidRoll = (typeof rn === 'number' || /^\d+$/.test(String(rn ?? '').trim()));
        const hasName = (name && String(name).trim() !== '' && !['TOTAL', 'AVERAGE', 'PERCENTAGE', 'GRADE'].includes(String(name).trim().toUpperCase()));
        const hasMarks = marks.some(v => v !== null && v !== undefined && String(v).trim() !== '');
        if(hasName || (hasValidRoll && hasMarks)){
          students.push({ rn: rn || (students.length + 1), name: hasName ? String(name).trim() : '-', marks: marks, grand: rows[r]?.[totalCol] ?? null, per: rows[r]?.[perCol] ?? null });
        }
      }
      if(!students.length) continue;
      const title = (rows[0]||[]).find(v=>typeof v==='string' && v.trim().length > 3) || 'BRAHMADEV PUBLIC SCHOOL SITAMARHI';
      const subtitleRaw = (rows[1]||[]).find(v=>typeof v==='string' && v.trim().length > 3) || 'FORMATIVE ASSESSMENT I 2026-27';
      const subtitleBase = subtitleRaw.replace(/Class\s*:\s*.*$/i, '').trim().replace(/;$/, '').trim();
      models.push({ examType: 'FA', sheetName, title: title.trim(), subtitleBase, subjects: finalSubjects, students });
    }
  }
  if(!models.length) throw new Error('No valid class result detected in sheet data.');
  return models;
}

function renderStudentCardSA(st, subj1, subj2, maxCols, stIdx){
  const hasRow2 = subj2.length > 0;
  const totalRowspan = hasRow2 ? 6 : 3;
  const colors = getThemeColors();

  let h = `<table class="student-card"><tbody>
    <tr>
      <td rowspan="${totalRowspan}" class="rn-box col-rn" style="background:#f1f5f9;color:${colors.rnTextColor}">
        <span class="rn-label" style="color:${colors.rnTextColor}">RN</span>${fmt(st.rn)}
      </td>
      <td rowspan="${totalRowspan}" class="name-box col-name" style="background:#f1f5f9;color:${colors.nameTextColor}">
        <span class="name-label" style="color:${colors.nameTextColor}">STUDENT NAME</span>${esc(st.name)}
      </td>`;
      
      let r1ColSum = 0;
      subj1.forEach((s, sIdx) => {
        const hc = getHeaderColorsForSubject(sIdx);
        h += `<th colspan="${s.colCount}" class="subj-head-1" style="background:${hc.headBg};color:${colors.headTextColor}">${esc(s.name)}</th>`;
        r1ColSum += s.colCount;
      });
      if(r1ColSum < maxCols){
        h += `<th colspan="${maxCols - r1ColSum}" class="subj-head-1" style="background:#e2e8f0;color:${colors.headTextColor}">-</th>`;
      }

      h += `<td rowspan="${totalRowspan}" class="grand-box col-summary-tot" style="background:${colors.grandBg};color:${colors.grandText}">
        <span class="grand-label" style="color:${colors.grandText}">GRAND<br>TOTAL</span>${fmt(st.grand)}
      </td>
      <td rowspan="${totalRowspan}" class="grand-box col-summary-per" style="background:${colors.grandBg};color:${colors.grandText}">
        <span class="grand-label" style="color:${colors.grandText}">PER<br>(%)</span>${fmt(st.per)}
      </td>
    </tr>
    <tr>`;
      subj1.forEach((s, sIdx) => {
        const hc = getHeaderColorsForSubject(sIdx);
        for(let k=0; k<s.colCount; k++){
          h += `<th class="subhead-cell-1" style="background:${hc.subBg};color:${colors.subheadTextColor}">${formatSubhead(s.subheads[k])}</th>`;
        }
      });
      if(r1ColSum < maxCols){
        for(let k=0; k<(maxCols - r1ColSum); k++) h += `<th class="subhead-cell-1" style="background:#f1f5f9;color:${colors.subheadTextColor}">-</th>`;
      }
    h += `</tr>
    <tr>`;
      subj1.forEach((s, idx) => {
        const b = st.blocks[idx] || [];
        for(let k=0; k<s.colCount; k++){
          const isTot = (k === s.colCount - 1);
          h += `<td class="mark-val ${isTot ? 'total-val' : ''}" style="color:${colors.marksTextColor}">${fmt(b[k])}</td>`;
        }
      });
      if(r1ColSum < maxCols){
        for(let k=0; k<(maxCols - r1ColSum); k++) h += `<td class="mark-val" style="color:${colors.marksTextColor}">-</td>`;
      }
    h += `</tr>`;

    if(hasRow2){
      let r2ColSum = 0;
      h += `<tr>`;
        subj2.forEach((s, idx2) => {
          const actualIdx = subj1.length + idx2;
          const hc = getHeaderColorsForSubject(actualIdx);
          h += `<th colspan="${s.colCount}" class="subj-head-2" style="background:${hc.headBg};color:${colors.headTextColor}">${esc(s.name)}</th>`;
          r2ColSum += s.colCount;
        });
        if(r2ColSum < maxCols){
          h += `<th colspan="${maxCols - r2ColSum}" class="subj-head-2" style="background:#e2e8f0;color:${colors.headTextColor}">-</th>`;
        }
      h += `</tr>
      <tr>`;
        subj2.forEach((s, idx2) => {
          const actualIdx = subj1.length + idx2;
          const hc = getHeaderColorsForSubject(actualIdx);
          for(let k=0; k<s.colCount; k++){
            h += `<th class="subhead-cell-2" style="background:${hc.subBg};color:${colors.subheadTextColor}">${formatSubhead(s.subheads[k])}</th>`;
          }
        });
        if(r2ColSum < maxCols){
          for(let k=0; k<(maxCols - r2ColSum); k++) h += `<th class="subhead-cell-2" style="background:#f1f5f9;color:${colors.subheadTextColor}">-</th>`;
        }
      h += `</tr>
      <tr>`;
        subj2.forEach((s, idx) => {
          const b = st.blocks[subj1.length + idx] || [];
          for(let k=0; k<s.colCount; k++){
            const isTot = (k === s.colCount - 1);
            h += `<td class="mark-val ${isTot ? 'total-val' : ''}" style="color:${colors.marksTextColor}">${fmt(b[k])}</td>`;
          }
        });
        if(r2ColSum < maxCols){
          for(let k=0; k<(maxCols - r2ColSum); k++) h += `<td class="mark-val" style="color:${colors.marksTextColor}">-</td>`;
        }
      h += `</tr>`;
    }

  h += `</tbody></table>`;
  return h;
}

function buildModelHtml(model, sheetIndex){
  const isFA = (model.examType === 'FA');
  const totalSubj = model.subjects.length;
  const colors = getThemeColors();
  let perPage = (perPageSelect.value === 'auto') ? (isFA ? 15 : 5) : parseInt(perPageSelect.value, 10);
  const totalPages = Math.ceil(model.students.length / perPage);
  let fullHtml = '';

  const activeTitle = editSchoolTitle.value.trim() || model.title;
  const baseSubtitle = editSchoolSubtitle.value.trim() || model.subtitleBase;

  if(isFA){
    for(let p=0; p<totalPages; p++){
      const slice = model.students.slice(p * perPage, (p + 1) * perPage);
      const isFirstPage = (p === 0);
      let pageHtml = `<div class="sheet">`;
      if(isFirstPage){
        pageHtml += `<div class="sheet-header"><div class="header-text-box"><div class="title" style="color:${colors.titleColor}">${esc(activeTitle)}</div><div class="subtitle" style="color:${colors.subtitleColor}">${esc(baseSubtitle)} ; Class : ${esc(model.sheetName)}</div></div></div>`;
      }
      pageHtml += `<div class="fa-table-wrap ${!isFirstPage ? 'fa-table-wrap-p2' : ''}"><table class="fa-table"><thead><tr><th class="fa-rn" style="width:9mm">ROLL</th><th class="fa-name" style="width:44mm">NAME OF STUDENTS</th>`;
      model.subjects.forEach((s, sIdx) => {
        const hc = getHeaderColorsForSubject(sIdx);
        pageHtml += `<th class="fa-head" style="background:${hc.headBg};color:${colors.headTextColor}">${esc(s.name)}<br><span style="font-size:calc(7.5px * var(--font-scale));font-weight:normal;color:${colors.subheadTextColor}">(20)</span></th>`;
      });
      pageHtml += `<th class="fa-summary" style="width:16mm;background:${colors.grandBg};color:${colors.grandText}">TOTAL</th><th class="fa-summary" style="width:22mm;background:${colors.grandBg};color:${colors.grandText}">PER (%)</th></tr></thead><tbody>`;

      slice.forEach((st) => {
        pageHtml += `<tr><td class="fa-rn-val" style="background:#f1f5f9;color:${colors.rnTextColor}">${fmt(st.rn)}</td><td class="fa-name-val" style="background:#f1f5f9;color:${colors.nameTextColor}">${esc(st.name)}</td>`;
        st.marks.forEach(m => { pageHtml += `<td class="fa-mark-val" style="color:${colors.marksTextColor}">${fmt(m)}</td>`; });
        pageHtml += `<td class="fa-grand-val" style="background:${colors.grandBg};color:${colors.grandText}">${fmt(st.grand)}</td><td class="fa-grand-val fa-per-val" style="background:${colors.grandBg};color:${colors.grandText}">${fmt(st.per)}</td></tr>`;
      });
      pageHtml += `</tbody></table></div><div class="sheet-footer"><span>Class: <b>${esc(model.sheetName)}</b> (${totalSubj} Subj)</span><span>A4 LANDSCAPE</span><span>Page <b>${p + 1}</b> of <b>${totalPages}</b></span></div></div>`;
      fullHtml += pageHtml;
    }
  } else {
    const isMultiRow = (totalSubj > 4);
    let splitCount = (totalSubj <= 4) ? totalSubj : Math.ceil(totalSubj / 2);
    const subj1 = model.subjects.slice(0, splitCount);
    const subj2 = model.subjects.slice(splitCount);
    const r1Cols = subj1.reduce((sum, s) => sum + s.colCount, 0);
    const r2Cols = subj2.reduce((sum, s) => sum + s.colCount, 0);
    const maxCols = Math.max(r1Cols, r2Cols);

    for(let p=0; p<totalPages; p++){
      const slice = model.students.slice(p * perPage, (p + 1) * perPage);
      const isFirstPage = (p === 0);
      let pageHtml = `<div class="sheet">`;
      if(isFirstPage){
        pageHtml += `<div class="sheet-header"><div class="header-text-box"><div class="title" style="color:${colors.titleColor}">${esc(activeTitle)}</div><div class="subtitle" style="color:${colors.subtitleColor}">${esc(baseSubtitle)} ; Class : ${esc(model.sheetName)}</div></div></div>`;
      }
      pageHtml += `<div class="student-container ${!isFirstPage ? 'student-container-p2' : ''}">`;
      slice.forEach((st, idx) => { pageHtml += renderStudentCardSA(st, subj1, subj2, maxCols, p * perPage + idx); });
      pageHtml += `</div><div class="sheet-footer"><span>Class: <b>${esc(model.sheetName)}</b> (${totalSubj} Subj)</span><span>A4 LANDSCAPE</span><span>Page <b>${p + 1}</b> of <b>${totalPages}</b></span></div></div>`;
      fullHtml += pageHtml;
    }
  }
  return fullHtml;
}

function generateCustomFileName(sheetName, subtitleText, loadedDocTitle){
  const cleanSheet = String(sheetName || '').trim();
  const fileStr = (loadedDocTitle || '') + ' ' + (subtitleText || '');
  let sec = ''; const secMatch = fileStr.match(/SEC(?:TION)?\s*[-_:]?\s*([A-Z])/i) || cleanSheet.match(/\b([A-Z])\b$/);
  if(secMatch) sec = `_SECTION_${secMatch[1].toUpperCase()}`;
  let exam = '_RESULT_EXAM';
  const cleanClass = cleanSheet.replace(/\s+/g,'_').toUpperCase();
  return `CLASS_${cleanClass}${sec}${exam}`.replace(/[<>:"/\\|?*]+/g, '_');
}

function blobToBase64(blob){
  return new Promise((resolve, reject) => {
    const reader = new FileReader(); reader.onloadend = () => resolve(reader.result.split(',')[1]); reader.onerror = reject; reader.readAsDataURL(blob);
  });
}
